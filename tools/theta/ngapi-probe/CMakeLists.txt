cmake_minimum_required(VERSION 3.24)
project(theta_ngapi_probe LANGUAGES CXX)
if(NOT DEFINED NGAPI_SOURCE_DIR OR NOT EXISTS "${NGAPI_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set NGAPI_SOURCE_DIR to the separately owned pinned NGAPI checkout")
endif()
set(NOGRAPHICSAPI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NOGRAPHICSAPI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${NGAPI_SOURCE_DIR}" ngapi)
add_executable(theta_ngapi_capabilities capabilities.cpp)
target_link_libraries(theta_ngapi_capabilities PRIVATE NoGraphicsAPI::NoGraphicsAPI)
add_executable(theta_ngapi_commands "${NGAPI_SOURCE_DIR}/tests/command_context_test.cpp")
target_link_libraries(theta_ngapi_commands PRIVATE NoGraphicsAPI::NoGraphicsAPI)
if(DEFINED THETA_SHADER_DIR)
    if(NOT EXISTS "${THETA_SHADER_DIR}/physical_readback_modules.hpp")
        message(FATAL_ERROR "Run compile_readback.py before configuring the Rust shader consumer")
    endif()
    add_executable(theta_ngapi_physical_readback physical_readback.cpp)
    target_include_directories(theta_ngapi_physical_readback PRIVATE "${THETA_SHADER_DIR}")
    target_link_libraries(theta_ngapi_physical_readback PRIVATE NoGraphicsAPI::NoGraphicsAPI)
    if(MSVC)
        target_compile_options(theta_ngapi_physical_readback PRIVATE /W4 /WX)
    endif()
endif()
if(DEFINED THETA_CUBE_SHADER_DIR)
    if(NOT EXISTS "${THETA_CUBE_SHADER_DIR}/native_heap_cube_modules.hpp")
        message(FATAL_ERROR "Compile native_heap_cube before configuring its consumer")
    endif()
    add_executable(theta_ngapi_native_heap_cube native_heap_cube.cpp)
    target_include_directories(theta_ngapi_native_heap_cube PRIVATE "${THETA_CUBE_SHADER_DIR}")
    target_link_libraries(theta_ngapi_native_heap_cube PRIVATE NoGraphicsAPI::NoGraphicsAPI)
    if(MSVC)
        target_compile_options(theta_ngapi_native_heap_cube PRIVATE /W4 /WX)
    endif()
endif()
if(DEFINED THETA_DIVERGENT_SHADER_DIR)
    if(NOT EXISTS "${THETA_DIVERGENT_SHADER_DIR}/native_heap_divergent_modules.hpp")
        message(FATAL_ERROR "Compile native_heap_divergent before configuring its consumer")
    endif()
    add_executable(theta_ngapi_native_heap_divergent native_heap_sample.cpp)
    target_compile_definitions(theta_ngapi_native_heap_divergent PRIVATE THETA_HEAP_DIVERGENT)
    target_include_directories(theta_ngapi_native_heap_divergent PRIVATE "${THETA_DIVERGENT_SHADER_DIR}")
    target_link_libraries(theta_ngapi_native_heap_divergent PRIVATE NoGraphicsAPI::NoGraphicsAPI)
    if(MSVC)
        target_compile_options(theta_ngapi_native_heap_divergent PRIVATE /W4 /WX)
    endif()
endif()
if(DEFINED THETA_HEAP_SHADER_DIR)
    if(NOT EXISTS "${THETA_HEAP_SHADER_DIR}/native_heap_sample_modules.hpp")
        message(FATAL_ERROR "Compile native_heap_sample before configuring its consumer")
    endif()
    add_executable(theta_ngapi_native_heap_sample native_heap_sample.cpp)
    target_include_directories(theta_ngapi_native_heap_sample PRIVATE "${THETA_HEAP_SHADER_DIR}")
    target_link_libraries(theta_ngapi_native_heap_sample PRIVATE NoGraphicsAPI::NoGraphicsAPI)
    if(MSVC)
        target_compile_options(theta_ngapi_native_heap_sample PRIVATE /W4 /WX)
    endif()
endif()

cmake_minimum_required(VERSION 3.24)
project(uvsr_launcher LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "UVSR Launcher requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(UVSR_LAUNCHER_SOURCE_IDENTITY "" CACHE STRING "source identity from the launcher build record")
set(UVSR_LAUNCHER_SOURCE_COMMIT "" CACHE STRING "exact source commit")
option(UVSR_LAUNCHER_PRODUCTION "clean production source boundary" OFF)
if(UVSR_LAUNCHER_SOURCE_IDENTITY STREQUAL "")
    message(FATAL_ERROR "use launcher/build.ps1 to establish source identity")
endif()
if(UVSR_LAUNCHER_PRODUCTION)
    find_package(Git REQUIRED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain=v1 --untracked-files=all --ignore-submodules=none
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.." OUTPUT_VARIABLE production_status RESULT_VARIABLE status_result OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.."
        OUTPUT_VARIABLE source_head RESULT_VARIABLE head_result OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT status_result EQUAL 0 OR NOT head_result EQUAL 0 OR NOT production_status STREQUAL "" OR
       NOT source_head STREQUAL UVSR_LAUNCHER_SOURCE_COMMIT OR NOT source_head STREQUAL UVSR_LAUNCHER_SOURCE_IDENTITY)
        message(FATAL_ERROR "production launcher requires a clean exact source identity")
    endif()
endif()
include(FetchContent)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib
    URL
        https://github.com/madler/zlib/releases/download/v1.3.2/zlib-1.3.2.tar.gz
        https://zlib.net/fossils/zlib-1.3.2.tar.gz
    URL_HASH SHA256=bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(zlib)

set(inventory_header "#pragma once\n#include <string_view>\nnamespace uvsr::launcher {\n")
foreach(inventory IN ITEMS runtime-shader-inventory runtime-asset-map)
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/${inventory}.def" rows)
    string(REPLACE "-" "_" variable "${inventory}")
    string(APPEND inventory_header "inline constexpr std::string_view ${variable}[] = {\n")
    foreach(row IN LISTS rows)
        string(REGEX REPLACE "[|].*$" "" path "${row}")
        string(APPEND inventory_header "\"${path}\",\n")
    endforeach()
    string(APPEND inventory_header "};\n")
endforeach()
string(APPEND inventory_header "}\n")
include(../cmake/RuntimeLicenseMappings.cmake)
set(license_header "namespace uvsr::launcher { inline constexpr std::string_view runtime_license_inventory[] = {\n")
while(UVSR_LICENSE_MAPPINGS)
    list(POP_FRONT UVSR_LICENSE_MAPPINGS source destination)
    string(APPEND license_header "\"bin/licenses/${destination}\",\n")
endwhile()
string(APPEND license_header "}; }\n")
string(APPEND inventory_header "${license_header}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/runtime_inventory.h" CONTENT "${inventory_header}")

add_library(uvsr_launcher_core STATIC native/platform.cpp native/trust.cpp native/download.cpp
    native/archive.cpp native/package.cpp native/process.cpp native/shell.cpp
    native/launcher_update.cpp native/installer.cpp native/cleanup.cpp
    ../src/settings_catalog.cpp ../src/settings_contract.cpp ../src/settings_value.cpp)
target_include_directories(uvsr_launcher_core PUBLIC native ../src ../tools "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(uvsr_launcher_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(uvsr_launcher_core PRIVATE /W4 /utf-8 /permissive- /constexpr:steps100000000)
target_link_libraries(uvsr_launcher_core PUBLIC bcrypt advapi32 shell32 ole32 uuid version winhttp zlibstatic)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/core.h" constants)
string(REGEX MATCH "LauncherVersion\\[\\] = \"([0-9.]+)\"" version_match "${constants}")
set(UVSR_LAUNCHER_VERSION "${CMAKE_MATCH_1}")
string(REGEX MATCH "LauncherSequence = ([0-9]+)" sequence_match "${constants}")
set(UVSR_LAUNCHER_SEQUENCE "${CMAKE_MATCH_1}")
if(NOT version_match OR NOT sequence_match)
    message(FATAL_ERROR "native launcher release identity is missing")
endif()
string(REPLACE "." "," UVSR_LAUNCHER_FILE_VERSION "${UVSR_LAUNCHER_VERSION}.0")
set(UVSR_LAUNCHER_PRODUCTION_TEXT "false")
if(UVSR_LAUNCHER_PRODUCTION)
    set(UVSR_LAUNCHER_PRODUCTION_TEXT "true")
endif()
configure_file(native/launcher.rc generated/launcher.rc @ONLY)
configure_file(native/launcher.manifest generated/launcher.manifest @ONLY)
add_executable(uvsr-launcher WIN32 native/main.cpp native/ui.cpp native/ui_support.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/generated/launcher.rc")
target_link_libraries(uvsr-launcher PRIVATE uvsr_launcher_core comctl32 user32 gdi32)
target_compile_options(uvsr-launcher PRIVATE /W4 /utf-8 /permissive-)
target_link_options(uvsr-launcher PRIVATE /DEBUG:NONE /MANIFEST:EMBED
    "/MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/launcher.manifest")
set(UVSR_LAUNCHER_BINARY_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin" CACHE PATH "launcher artifact output")
set_target_properties(uvsr-launcher PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${UVSR_LAUNCHER_BINARY_DIRECTORY}")

include(CTest)
if(BUILD_TESTING)
    # resource fixtures model old and current packages without a managed runtime.
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../src/settings_snapshot_schema_versions.def" registry REGEX "^UVSR_SETTINGS_SNAPSHOT_SCHEMA_VERSION")
    list(GET registry -1 current_schema)
    foreach(kind IN ITEMS current legacy)
        if(kind STREQUAL "legacy")
            set(row "UVSR_SETTINGS_SNAPSHOT_SCHEMA_VERSION(0x000bu, 0xdf2b9ab2a2c8d65eull, 0x4415dc93c0051f0cull)")
        else()
            set(row "${current_schema}")
        endif()
        string(REGEX MATCH "0x([0-9a-f]+)ull, 0x([0-9a-f]+)ull" match "${row}")
        set(FIXTURE_HASH "${CMAKE_MATCH_1}${CMAKE_MATCH_2}")
        set(FIXTURE_VERSION "")
        foreach(offset RANGE 0 12 4)
            string(SUBSTRING "${FIXTURE_HASH}" ${offset} 4 part)
            math(EXPR part "0x${part}")
            list(APPEND FIXTURE_VERSION "${part}")
        endforeach()
        list(JOIN FIXTURE_VERSION "." FIXTURE_VERSION)
        string(REPLACE "." "," FIXTURE_VERSION_COMMAS "${FIXTURE_VERSION}")
        configure_file(tests/native/engine.rc "generated/fixture-${kind}.rc" @ONLY)
        add_executable(fixture_engine_${kind} tests/native/fixture.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/fixture-${kind}.rc")
        set_target_properties(fixture_engine_${kind} PROPERTIES OUTPUT_NAME "uvsr-engine" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${kind}")
    endforeach()
    set(current_launcher_version "${UVSR_LAUNCHER_VERSION}")
    set(current_launcher_sequence "${UVSR_LAUNCHER_SEQUENCE}")
    foreach(kind IN ITEMS current legacy newer)
        if(kind STREQUAL "legacy")
            set(UVSR_LAUNCHER_VERSION "1.2.0")
            set(UVSR_LAUNCHER_SEQUENCE 16)
        elseif(kind STREQUAL "newer")
            set(UVSR_LAUNCHER_VERSION "1.4.0")
            math(EXPR UVSR_LAUNCHER_SEQUENCE "${current_launcher_sequence} + 1")
        else()
            set(UVSR_LAUNCHER_VERSION "${current_launcher_version}")
            set(UVSR_LAUNCHER_SEQUENCE "${current_launcher_sequence}")
        endif()
        set(UVSR_LAUNCHER_SOURCE_IDENTITY "1111111111111111111111111111111111111111")
        set(UVSR_LAUNCHER_SOURCE_COMMIT "${UVSR_LAUNCHER_SOURCE_IDENTITY}")
        string(REPLACE "." "," UVSR_LAUNCHER_FILE_VERSION "${UVSR_LAUNCHER_VERSION}.0")
        configure_file(native/launcher.rc "generated/fixture-launcher-${kind}.rc" @ONLY)
        add_executable(fixture_launcher_${kind} tests/native/fixture.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/fixture-launcher-${kind}.rc")
        set_target_properties(fixture_launcher_${kind} PROPERTIES OUTPUT_NAME "uvsr-launcher" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/launcher-${kind}")
    endforeach()
    add_executable(uvsr_launcher_tests tests/native/contracts.cpp tests/native/fixtures.cpp tests/native/system_proof.cpp)
    target_link_libraries(uvsr_launcher_tests PRIVATE uvsr_launcher_core crypt32 user32 gdi32)
    # retained package paths need the same long-path support as the launcher.
    target_link_options(uvsr_launcher_tests PRIVATE
        "/MANIFEST:EMBED"
        "/MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/launcher.manifest")
    target_compile_options(uvsr_launcher_tests PRIVATE /W4 /utf-8 /permissive-)
    target_compile_definitions(uvsr_launcher_tests PRIVATE
        "UVSR_TEST_SOURCE=L\"${CMAKE_CURRENT_SOURCE_DIR}\""
        "UVSR_ENGINE_FIXTURE=L\"$<TARGET_FILE:fixture_engine_current>\""
        "UVSR_LEGACY_ENGINE_FIXTURE=L\"$<TARGET_FILE:fixture_engine_legacy>\""
        "UVSR_LAUNCHER_FIXTURE=L\"$<TARGET_FILE:fixture_launcher_current>\""
        "UVSR_LEGACY_LAUNCHER_FIXTURE=L\"$<TARGET_FILE:fixture_launcher_legacy>\""
        "UVSR_NEWER_LAUNCHER_FIXTURE=L\"$<TARGET_FILE:fixture_launcher_newer>\"")
    add_dependencies(uvsr_launcher_tests fixture_engine_current fixture_engine_legacy fixture_launcher_current fixture_launcher_legacy fixture_launcher_newer)
    add_test(NAME launcher_pure COMMAND uvsr_launcher_tests --pure)
    option(UVSR_LAUNCHER_SYSTEM_TESTS "enable registry, shell and process tests on a suitable validation host" OFF)
    if(UVSR_LAUNCHER_SYSTEM_TESTS)
        add_test(NAME launcher_system_services COMMAND uvsr_launcher_tests --system-services)
        add_test(NAME launcher_runtime COMMAND uvsr_launcher_tests --runtime)
    endif()
endif()
